Building a Python CLI with Click

From zero to a published PyPI package: commands, subcommands, options, testing, packaging, and automation. No fluff.

What You'll Build

  • A CLI with nested subcommands (mytool init, mytool deploy staging)
  • Typed options with validation and defaults
  • Config file support (YAML/TOML)
  • Unit tests with pytest + click.testing.CliRunner
  • Packaging with pyproject.toml + uv/pip
  • GitHub Actions CI → PyPI release on tag push

"""Main CLI entry point."""
from __future__ import annotations

import sys
from pathlib import Path
from typing import Optional

import click
from rich.console import Console
from rich.table import Table

from .config import Config, load_config
from .commands import init, deploy, status

console = Console(stderr=True)

# ──────────────────────────────────────────────
# Context object passed to all subcommands
# ──────────────────────────────────────────────
class Context:
    def __init__(self, config: Config, verbose: bool = False):
        self.config = config
        self.verbose = verbose


# ──────────────────────────────────────────────
# Root command group
# ──────────────────────────────────────────────
@click.group(
    name="click-cli-demo",
    help="Demo CLI built with Click — deploy, init, and check status.",
    context_settings={"help_option_names": ["-h", "--help"]},
)
@click.option(
    "-c", "--config",
    type=click.Path(exists=True, dir_okay=False, path_type=Path),
    default=None,
    help="Path to config file (YAML/TOML). Default: ./config.yaml or ~/.config/click-cli-demo/config.yaml",
)
@click.option("-v", "--verbose", is_flag=True, help="Enable verbose output")
@click.version_option(version="0.1.0", prog_name="click-cli-demo")
@click.pass_context
def cli(ctx: click.Context, config: Optional[Path], verbose: bool) -> None:
    """Root command — loads config and stores Context for subcommands."""
    cfg = load_config(config)
    ctx.obj = Context(config=cfg, verbose=verbose)
    if verbose:
        console.print(f"[dim]Loaded config from: {cfg.source_path}[/dim]")


# ──────────────────────────────────────────────
# Subcommands (imported from commands/)
# ──────────────────────────────────────────────
cli.add_command(init.init_cmd)
cli.add_command(deploy.deploy_cmd)
cli.add_command(status.status_cmd)


# ──────────────────────────────────────────────
# Entry point for `python -m cli.main`
# ──────────────────────────────────────────────
def main() -> int:
    try:
        cli(obj={})
        return 0
    except click.ClickException as e:
        e.show()
        return e.exit_code
    except Exception as e:  # pragma: no cover
        console.print(f"[red]Unexpected error:[/red] {e}")
        if "--verbose" in sys.argv:
            import traceback
            console.print(traceback.format_exc())
        return 1


if __name__ == "__main__":
    sys.exit(main())

Create src/cli/config.py

"""Configuration loading with YAML/TOML support."""
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Optional

import yaml

try:
    import tomllib  # py311+
except ModuleNotFoundError:  # pragma: no cover
    import tomli as tomllib


@dataclass
class DeployTarget:
    name: str
    host: str
    user: str = "deploy"
    path: str = "/var/www/app"
    env_file: Optional[str] = None
    pre_deploy: list[str] = field(default_factory=list)
    post_deploy: list[str] = field(default_factory=list)


@dataclass
class Config:
    project_name: str = "my-app"
    default_target: str = "staging"
    targets: dict[str, DeployTarget] = field(default_factory=dict)
    source_path: Optional[Path] = None

    def get_target(self, name: str) -> DeployTarget:
        if name not in self.targets:
            raise KeyError(f"Target '{name}' not defined in config. Available: {list(self.targets.keys())}")
        return self.targets[name]


DEFAULT_CONFIG_PATHS = [
    Path.cwd() / "config.yaml",
    Path.cwd() / "config.yml",
    Path.cwd() / "config.toml",
    Path.home() / ".config" / "click-cli-demo" / "config.yaml",
]


def load_config(explicit_path: Optional[Path] = None) -> Config:
    """Load config from explicit path, or search default locations."""
    search_paths = [explicit_path] if explicit_path else DEFAULT_CONFIG_PATHS

    for path in search_paths:
        if path and path.exists():
            return _parse_config(path)

    # No config found — return defaults
    return Config()


def _parse_config(path: Path) -> Config:
    text = path.read_text(encoding="utf-8")
    if path.suffix in {".yaml", ".yml"}:
        data = yaml.safe_load(text) or {}
    elif path.suffix == ".toml":
        data = tomllib.loads(text)
    else:
        raise ValueError(f"Unsupported config format: {path.suffix}. Use .yaml, .yml, or .toml")

    targets = {}
    for name, tgt in data.get("targets", {}).items():
        targets[name] = DeployTarget(name=name, **tgt)

    return Config(
        project_name=data.get("project_name", "my-app"),
        default_target=data.get("default_target", "staging"),
        targets=targets,
        source_path=path,
    )

"""Init command: scaffold a new project."""
from __future__ import annotations

from pathlib import Path

import click
from rich.console import Console
from rich.prompt import Prompt, Confirm

console = Console()

TEMPLATE_CONFIG = """# click-cli-demo configuration
project_name: "{project_name}"
default_target: "staging"

targets:
  staging:
    host: "staging.example.com"
    user: "deploy"
    path: "/var/www/{project_name}"
    env_file: ".env.staging"
    pre_deploy:
      - "echo 'Running migrations...'"
    post_deploy:
      - "systemctl reload nginx"

  production:
    host: "prod.example.com"
    user: "deploy"
    path: "/var/www/{project_name}"
    env_file: ".env.production"
    pre_deploy:
      - "echo 'Running migrations...'"
      - "echo 'Backup database...'"
    post_deploy:
      - "systemctl reload nginx"
      - "echo 'Health check...'"
"""

@click.command(
    name="init",
    help="Initialize a new project with config template.",
)
@click.argument("project_name", required=False)
@click.option("-f", "--force", is_flag=True, help="Overwrite existing config")
@click.pass_obj
def init_cmd(obj, project_name: str | None, force: bool) -> None:
    """Scaffold config.yaml in current directory."""
    config_path = Path.cwd() / "config.yaml"

    if config_path.exists() and not force:
        if not Confirm.ask(f"[yellow]{config_path} exists. Overwrite?[/yellow]"):
            console.print("[dim]Aborted.[/dim]")
            return

    if not project_name:
        project_name = Prompt.ask("Project name", default=Path.cwd().name)

    content = TEMPLATE_CONFIG.format(project_name=project_name)
    config_path.write_text(content, encoding="utf-8")
    console.print(f"[green]✓[/green] Created {config_path}")
    console.print("[dim]Edit it to match your deployment targets.[/dim]")


# Export for main.py
__all__ = ["init_cmd"]

"""Deploy command: push code to a target server."""
from __future__ import annotations

import subprocess
from pathlib import Path

import click
from rich.console import Console
from rich.progress import Progress, SpinnerColumn, TextColumn

from ..config import Context

console = Console()

@click.command(
    name="deploy",
    help="Deploy to a target environment.",
)
@click.argument("target", required=False)
@click.option("--dry-run", is_flag=True, help="Show commands without executing")
@click.option("--skip-pre", is_flag=True, help="Skip pre-deploy hooks")
@click.option("--skip-post", is_flag=True, help="Skip post-deploy hooks")
@click.pass_obj
def deploy_cmd(obj: Context, target: str | None, dry_run: bool, skip_pre: bool, skip_post: bool) -> None:
    """Deploy the project to TARGET (default: config.default_target)."""
    cfg = obj.config
    target_name = target or cfg.default_target

    try:
        tgt = cfg.get_target(target_name)
    except KeyError as e:
        raise click.ClickException(str(e))

    console.print(f"[bold]Deploying[/bold] [cyan]{cfg.project_name}[/cyan] → [magenta]{tgt.name}[/magenta]")

    # Build rsync command
    rsync_cmd = [
        "rsync", "-avz", "--delete",
        "--exclude=.git", "--exclude=__pycache__", "--exclude=.venv",
        f"{Path.cwd()}/", f"{tgt.user}@{tgt.host}:{tgt.path}/",
    ]

    if dry_run:
        console.print("[dim]DRY RUN — would execute:[/dim]")
        console.print("  " + " ".join(rsync_cmd))
        return

    # Pre-deploy hooks
    if not skip_pre and tgt.pre_deploy:
        console.print("[dim]Running pre-deploy hooks...[/dim]")
        for hook in tgt.pre_deploy:
            _run_remote(tgt, hook, console)

    # Sync files
    with Progress(
        SpinnerColumn(), TextColumn("[progress.description]{task.description}"),
        console=console, transient=True,
    ) as progress:
        task = progress.add_task("Syncing files...", total=None)
        result = subprocess.run(rsync_cmd, capture_output=True, text=True)
        if result.returncode != 0:
            raise click.ClickException(f"rsync failed:\n{result.stderr}")

    # Post-deploy hooks
    if not skip_post and tgt.post_deploy:
        console.print("[dim]Running post-deploy hooks...[/dim]")
        for hook in tgt.post_deploy:
            _run_remote(tgt, hook, console)

    console.print(f"[green]✓[/green] Deployed to {tgt.name} ({tgt.host})")


def _run_remote(tgt, cmd: str, console: Console) -> None:
    full_cmd = ["ssh", f"{tgt.user}@{tgt.host}", cmd]
    result = subprocess.run(full_cmd, capture_output=True, text=True)
    if result.returncode != 0:
        console.print(f"[red]Hook failed:[/red] {cmd}\n{result.stderr}")
    elif obj.verbose:
        console.print(f"[dim]  {cmd}[/dim]")


__all__ = ["deploy_cmd"]

Create src/cli/commands/status.py

"""Status command: show config and target info."""
from __future__ import annotations

import click
from rich.console import Console
from rich.table import Table

from ..config import Context

console = Console()

@click.command(
    name="status",
    help="Show current configuration and deployment targets.",
)
@click.pass_obj
def status_cmd(obj: Context) -> None:
    """Display loaded config and available targets."""
    cfg = obj.config

    console.print(f"[bold]Project:[/bold] {cfg.project_name}")
    console.print(f"[bold]Default target:[/bold] {cfg.default_target}")
    if cfg.source_path:
        console.print(f"[bold]Config source:[/bold] {cfg.source_path}")
    else:
        console.print("[yellow]No config file loaded (using defaults)[/yellow]")

    if not cfg.targets:
        console.print("\n[dim]No deployment targets configured. Run `click-cli-demo init` to create one.[/dim]")
        return

    table = Table(title="Deployment Targets")
    table.add_column("Name", style="cyan")
    table.add_column("Host", style="magenta")
    table.add_column("User", style="green")
    table.add_column("Path", style="blue")
    table.add_column("Env File", style="yellow")

    for name, tgt in cfg.targets.items():
        marker = " ← default" if name == cfg.default_target else ""
        table.add_row(
            f"{name}{marker}",
            tgt.host,
            tgt.user,
            tgt.path,
            tgt.env_file or "—",
        )

    console.print()
    console.print(table)


__all__ = ["status_cmd"]

"""CLI tests using click.testing.CliRunner."""
from __future__ import annotations

import tempfile
from pathlib import Path

import pytest
from click.testing import CliRunner

from cli.main import cli


runner = CliRunner()


def test_help() -> None:
    result = runner.invoke(cli, ["--help"])
    assert result.exit_code == 0
    assert "click-cli-demo" in result.output
    assert "deploy" in result.output
    assert "init" in result.output
    assert "status" in result.output


def test_version() -> None:
    result = runner.invoke(cli, ["--version"])
    assert result.exit_code == 0
    assert "0.1.0" in result.output


def test_status_no_config() -> None:
    """status works without config file."""
    with runner.isolated_filesystem():
        result = runner.invoke(cli, ["status"])
        assert result.exit_code == 0
        assert "No config file loaded" in result.output


def test_init_creates_config() -> None:
    """init command creates config.yaml."""
    with runner.isolated_filesystem():
        result = runner.invoke(cli, ["init", "test-project", "--force"])
        assert result.exit_code == 0
        assert "Created config.yaml" in result.output

        config_path = Path("config.yaml")
        assert config_path.exists()
        content = config_path.read_text()
        assert "test-project" in content
        assert "staging" in content
        assert "production" in content


def test_deploy_requires_target() -> None:
    """deploy fails if no target configured."""
    with runner.isolated_filesystem():
        # Create minimal config without targets
        Path("config.yaml").write_text("project_name: test\n")
        result = runner.invoke(cli, ["deploy"])
        assert result.exit_code != 0
        assert "not defined" in result.output or "not configured" in result.output


def test_config_loads_targets() -> None:
    """Config parsing works with YAML."""
    from cli.config import load_config

    with runner.isolated_filesystem():
        config = """
project_name: myapp
default_target: staging
targets:
  staging:
    host: staging.example.com
    user: deploy
    path: /var/www/myapp
"""
        Path("config.yaml").write_text(config)
        cfg = load_config(Path("config.yaml"))
        assert cfg.project_name == "myapp"
        assert cfg.default_target == "staging"
        assert "staging" in cfg.targets
        assert cfg.targets["staging"].host == "staging.example.com"


if __name__ == "__main__":
    pytest.main([__file__, "-v"])

      - name: Run tests
        run: uv run pytest -v
      - name: Type check
        run: uv run mypy src
      - name: Lint
        run: uv run ruff check src tests

  build:
    needs: test
    runs-on: ubuntu-latest
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    steps:
      - uses: actions/checkout@v4
      - name: Install uv

🔐 Trusted Publishing (recommended)

Skip API tokens entirely. In PyPI project settings → "Add publisher" → GitHub → your repo → workflow name release.yml. Then remove password from the workflow.
